# ============================================================================
# CMakeLists.txt — CNN 手写汉字识别项目的构建配置
# ============================================================================
#
# CMake 是什么？
#   一个跨平台的 C++ 构建工具。你告诉它"源文件有哪些、用什么编译器标准"，
#   它生成对应的 Makefile（Linux/Mac）或 Visual Studio 工程（Windows）。
#   然后你用 cmake --build . 一条命令就能编译。
#
# ============================================================================
# 项目基本信息
# ============================================================================

# CMake 最低版本要求：3.20 以上才有我们需要的功能
cmake_minimum_required(VERSION 3.20)

# 项目名：cnn_hanzi（CNN 汉字识别）
# 这个名字同时决定了生成的可执行文件名：cnn_hanzi.exe
project(cnn_hanzi)

# ============================================================================
# 编译器设置
# ============================================================================

# 使用 C++17 标准（需要 if constexpr、std::optional 等特性）
set(CMAKE_CXX_STANDARD 17)

# REQUIRED：如果编译器不支持 C++17，直接报错，不要悄悄降级
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# 编译选项：
#   -Wall     → 开启所有警告（帮你发现潜在 bug）
#   -O2       → 二级优化（平衡编译速度和运行速度）
#   以后如果要调试可以改成 -O0 -g（关优化，加调试符号）
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall -O2")

# ============================================================================
# 头文件路径
# ============================================================================

# 让编译器知道去 include/ 目录找头文件
# 这样在 .cpp 里写 #include "matrix.h" 就能找到 include/matrix.h
include_directories(include)

# ============================================================================
# 可执行文件定义
# ============================================================================

# 告诉 CMake：把这些 .cpp 文件编译链接成一个可执行文件 cnn_hanzi
# 每新增一个 .cpp 文件，都要加到这里，否则链接时找不到符号
add_executable(cnn_hanzi
    main.cpp
    src/matrix.cpp
    src/tensor.cpp
)
